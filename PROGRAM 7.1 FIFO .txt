#include <stdio.h>
int fr[3];
void display() {
    for(int i = 0; i < 3; i++) {
        if(fr[i] != -1)
            printf("%d", fr[i]);
        else
            printf("-");
    }
}
int main() {
    int i, j;
    int page[12] = {2, 3, 2, 1, 5, 2, 4, 5, 3, 2, 5, 2};
    int frsize = 3;
    int top = 0;
    int pf = 0;
    for(i = 0; i < frsize; i++) {
        fr[i] = -1;
    }
    printf("Page\tFrames\t\tPage Fault\n");
    for(j = 0; j < 12; j++) {
        int flag1 = 0;
        int flag2 = 0;
        for(i = 0; i < frsize; i++) {
            if(fr[i] == page[j]) {
                flag1 = 1;
                break;
            }
        }
        if(flag1 == 0) {
            for(i = 0; i < frsize; i++) {
                if(fr[i] == -1) {
                    fr[i] = page[j];
                    pf++;
                    flag2 = 1;
                    break;
                }
            }
            if(flag1 == 0 && flag2 == 0) {
                fr[top] = page[j];
                top = (top + 1) % frsize;
                pf++;
            }
        }
        printf("%d\t", page[j]);
        display();
        if(flag1 == 0)
            printf("\tYes\n");
        else
            printf("\tNo\n");
    }
    printf("\nTotal Page Faults: %d\n", pf);
    return 0;
}
