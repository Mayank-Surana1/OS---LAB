#include <stdio.h>
int fr[3];
void displayFrames(char out[]) {
    int i;
    for(i = 0; i < 3; i++) {
        if(fr[i] == -1)
            out[i] = '-';
        else
            out[i] = fr[i] + '0';
    }
}
int main() {
    int p[12] = {2,3,2,1,5,2,4,5,3,2,5,2};
    int i,j,k,l;
    int s[3];
    int index = 0;
    int flag1, flag2;
    int pf = 0;
    int frsize = 3;
    char frameStr[4];
    for(i = 0; i < frsize; i++) {
        fr[i] = -1;
    }
    printf("Page\tFrames\t\tPage Fault\n");
    for(j = 0; j < 12; j++) {
        flag1 = 0; flag2 = 0;
        for(i = 0; i < frsize; i++) {
            if(fr[i] == p[j]) {
                flag1 = 1; flag2 = 1;
                break;
            }
        }
        if(flag1 == 0) {
            for(i = 0; i < frsize; i++) {
                if(fr[i] == -1) {
                    fr[i] = p[j];
                    flag2 = 1;
                    pf++;
                    break;
                }
            }
        }
        if(flag2 == 0) {
            for(i = 0; i < frsize; i++) {
                s[i] = 0;
            }
            for(k = j - 1, l = 1; l <= frsize - 1; l++, k--) {
                for(i = 0; i < frsize; i++) {
                    if(fr[i] == p[k]) {
                        s[i] = 1;
                    }
                }
            }
            for(i = 0; i < frsize; i++) {
                if(s[i] == 0) {
                    index = i;
                    break;
                }
            }
            fr[index] = p[j];
            pf++;
        }
        displayFrames(frameStr);
        if(flag1 == 1)
            printf("%d\t%s\tNo\n", p[j], frameStr);
        else
            printf("%d\t%s\tYes\n", p[j], frameStr);
    }
    printf("\nTotal Page Faults: %d\n", pf);
    return 0;
}
