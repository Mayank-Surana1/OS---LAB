#include <stdio.h>
#define MAX 25

int main() {
    int frag[MAX], b[MAX], f[MAX], i, j, nb, nf, temp, lowest;
    int bf[MAX] = {0}, ff[MAX] = {0};
    printf("Enter the number of blocks: ");
    scanf("%d", &nb);
    printf("Enter the number of files: ");
    scanf("%d", &nf);
    printf("\nEnter the size of the blocks:\n");
    for(i = 1; i <= nb; i++) {
        printf("Block %d: ", i);
        scanf("%d", &b[i]);
    }
    printf("\nEnter the size of the files:\n");
    for(i = 1; i <= nf; i++) {
        printf("File %d: ", i);
        scanf("%d", &f[i]);
    }
    for(i = 1; i <= nf; i++) {
        lowest = 10000;
        for(j = 1; j <= nb; j++) {
            if(bf[j] != 1) {
                temp = b[j] - f[i];
                if(temp >= 0 && temp < lowest) {
                    ff[i] = j;
                    frag[i] = temp;
                    lowest = temp;
                }
            }
        }
        if(ff[i] != 0)
            bf[ff[i]] = 1;
    }
    printf("\nFile No\tFile Size\tBlock No\tBlock Size\tFragment\n");
    for(i = 1; i <= nf; i++) {
        if(ff[i] != 0)
            printf("%d\t%d\t%d\t%d\t%d\n", i, f[i], ff[i], b[ff[i]], frag[i]);
        else
            printf("%d\t%d\t\tNot Allocated\n", i, f[i]);
    }
    return 0;
}
