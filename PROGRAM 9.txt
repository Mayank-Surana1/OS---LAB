#include <stdio.h>
#include <stdlib.h>

int main() {
    int f[50], p, i, j, k, a, st, len, n, c;
    for (i = 0; i < 50; i++)
        f[i] = 0;
    printf("Enter how many blocks are already allocated: ");
    scanf("%d", &p);
    printf("Enter the block numbers that are already allocated: ");
    for (i = 0; i < p; i++) {
        scanf("%d", &a);
        f[a] = 1;
    }
    do {
        printf("\nEnter the starting index block and length: ");
        scanf("%d%d",&st,&len);
        k = len;
        for (j = st; j < (k + st); j++) {
            if (f[j] == 0) {
                f[j] = 1;
                printf("%d -> Allocated\n", j);
            } else {
                printf("%d -> Already allocated\n", j);
                k++;
            }
        }
        printf("\nDo you want to enter another file? (Yes - 1 / No - 0): ");
        scanf("%d",&c);
    } while (c == 1);
    return 0;
}
