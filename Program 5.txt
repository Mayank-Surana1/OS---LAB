#include<stdio.h>
void main() {
char job[10][10];
int time[10], avail, tem[10], temp[10], safe[10];
int ind = 0, i, j, q, n, t;
printf("Enter number of jobs: ");
scanf("%d", &n);
for(i = 0; i < n; i++) {
printf("Enter job name and execution time: ");
scanf("%s %d", job[i], &time[i]);
}
printf("Enter available resources: ");
scanf("%d", &avail);
for(i = 0; i < n; i++) {
temp[i] = time[i];
tem[i] = i;
}
for(i = 0; i < n - 1; i++) {
for(j = i + 1; j < n; j++) {
if(temp[i] > temp[j]) {
t = temp[i];
temp[i] = temp[j];
temp[j] = t;
t = tem[i];
tem[i] = tem[j];
tem[j] = t;
}
}
}
printf("\nSafe sequence: ");
for(i = 0; i < n; i++) {
q = tem[i];
if(time[q] <= avail) {
safe[ind] = q;
avail -= time[q];
printf("%s ", job[safe[ind]]);
ind++;
} else {
printf("\nNo safe sequence possible for %s\n", job[q]);
}
}
printf("\nFinal safe jobs list:\n");
for(i = 0; i < ind; i++) {
printf("%s %d\n", job[safe[i]], time[safe[i]]);
}
}
